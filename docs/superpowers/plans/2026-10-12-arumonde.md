```js
// 名前の表記ゆれを畳む: NFKC → 空白除去 → カタカナをひらがな → 小文字
export function fold(name) {
  return String(name ?? '')
    .normalize('NFKC')
    .replace(/\s+/g, '')
    .replace(/[ァ-ヶ]/g, (c) => String.fromCharCode(c.charCodeAt(0) - 0x60))
    .toLowerCase()
}
```

```js
import { KINDS, SHELVES, findPreset, canonicalKey, shelfFor } from './presets.js'

const defaultId = () => crypto.randomUUID()

export function findItem(items, name) {
  const key = canonicalKey(name)
  return items.find((it) => canonicalKey(it.name) === key) ?? null
}

// 追加。既にあれば何もしない（切らしていれば戻す）
export function addItem(items, { name, kind, shelf } = {}, { now = Date.now(), makeId = defaultId } = {}) {
  const trimmed = String(name ?? '').trim()
  if (!trimmed) return { items, item: null, status: 'empty' }
  const existing = findItem(items, trimmed)
  if (existing) {
    if (!existing.outOfStock) return { items, item: existing, status: 'exists' }
    const item = { ...existing, outOfStock: false }
    return { items: items.map((it) => (it.id === item.id ? item : it)), item, status: 'restored' }
  }
  const preset = findPreset(trimmed)
  if (!preset && !KINDS.includes(kind)) return { items, item: null, status: 'invalid' }
  const k = preset ? preset.kind : kind
  const item = {
    id: makeId(),
    name: preset ? preset.name : trimmed,
    kind: k,
    shelf: preset ? preset.shelf : shelfFor(k, shelf),
    outOfStock: false,
    addedAt: now,
  }
  return { items: [...items, item], item, status: 'added' }
}

// 追加画面のタップ: 未登録→追加、在庫あり→削除、切らした→復活
export function toggleRegistered(items, entry, opts) {
  const existing = findItem(items, entry.name)
  if (existing && !existing.outOfStock) return removeItem(items, existing.id)
  return addItem(items, entry, opts).items
}

export function toggleOut(items, id) {
  return items.map((it) => (it.id === id ? { ...it, outOfStock: !it.outOfStock } : it))
}

export function removeItem(items, id) {
  return items.filter((it) => it.id !== id)
}

export function inStock(items, kind) {
  return items.filter((it) => !it.outOfStock && it.kind === kind)
}

export function groupByShelf(items) {
  return SHELVES.map((shelf) => ({ shelf, items: items.filter((it) => it.shelf === shelf.id) })).filter(
    (g) => g.items.length,
  )
}
```

```js
export const SERVINGS = [1, 2, 3, 4]
export const TIMES = [
  { id: 15, label: '15分以内' },
  { id: 30, label: '30分以内' },
]
export const MOODS = [
  { id: 'light', label: 'さっぱり', adj: 'さっぱりした' },
  { id: 'hearty', label: 'がっつり', adj: 'がっつりした' },
  { id: 'warm', label: 'あったか', adj: '体があたたまる' },
  { id: 'healthy', label: 'ヘルシー', adj: 'ヘルシーな' },
]
export const MAX_ENCODED = 6000
export const CLOSING =
  'これ以外の材料を使う場合は「要買い足し」と明記してください。各案に材料と分量、手順を簡潔にお願いします。'

function requestLine({ starred, servings, time, moods }) {
  const parts = []
  if (starred.length) parts.push(`${starred.join('と')}を必ず使った`)
  const t = TIMES.find((x) => x.id === time)
  if (servings && t) parts.push(`${servings}人分・${t.label}で作れる`)
  else if (servings) parts.push(`${servings}人分の`)
  else if (t) parts.push(`${t.label}で作れる`)
  const adj = MOODS.filter((m) => moods.includes(m.id)).map((m) => m.adj)
  if (adj.length) parts.push(adj.join('、'))
  return `${parts.join('、')}晩ごはんのレシピを3案ください。`
}

function compose(input, others, seasonings, cut) {
  const lines = [requestLine(input)]
  if (!input.starred.length && !input.others.length) {
    lines.push('食材の手持ちはないので、調味料だけで作れるもの、または少しの買い足しで作れるものでお願いします。')
  }
  if (others.length || cut.others) {
    const label = input.starred.length ? '他に使ってよい食材' : '使ってよい食材'
    lines.push(`${label}：${[...others, ...(cut.others ? ['ほか'] : [])].join('、')}`)
  }
  if (seasonings.length || cut.seasonings) {
    lines.push(`使ってよい調味料：${[...seasonings, ...(cut.seasonings ? ['ほか基本調味料'] : [])].join('、')}`)
  }
  lines.push(CLOSING)
  return lines.join('\n')
}

export function buildPrompt(input = {}, { maxEncoded = MAX_ENCODED } = {}) {
  const inp = { starred: [], others: [], seasonings: [], servings: null, time: null, moods: [], ...input }
  const others = [...inp.others]
  const seasonings = [...inp.seasonings]
  const cut = { others: false, seasonings: false }
  let text = compose(inp, others, seasonings, cut)
  // URL に載る長さに収める。★と条件は削らない
  while (encodeURIComponent(text).length > maxEncoded) {
    if (seasonings.length) {
      seasonings.pop()
      cut.seasonings = true
    } else if (others.length) {
      others.pop()
      cut.others = true
    } else break
    text = compose(inp, others, seasonings, cut)
  }
  return text
}

export function buildKeywords({ starred = [], others = [] } = {}) {
  return starred.length ? [...starred] : others.slice(0, 3)
}
```

```js
import { KINDS, shelfFor } from './presets.js'
import { SERVINGS, TIMES, MOODS } from './prompt.js'

export const KEY = 'arumonde:v1'

export function defaultState() {
  return { items: [], prefs: { servings: 2, time: null, moods: [] } }
}

function cleanItem(it) {
  if (!it || typeof it.id !== 'string' || !it.id) return null
  if (typeof it.name !== 'string' || !it.name.trim()) return null
  if (!KINDS.includes(it.kind)) return null
  return {
    id: it.id,
    name: it.name.trim(),
    kind: it.kind,
    shelf: shelfFor(it.kind, it.shelf),
    outOfStock: Boolean(it.outOfStock),
    addedAt: Number.isFinite(it.addedAt) ? it.addedAt : 0,
  }
}

function cleanPrefs(p = {}) {
  const d = defaultState().prefs
  return {
    servings: p.servings === null ? null : SERVINGS.includes(p.servings) ? p.servings : d.servings,
    time: TIMES.some((t) => t.id === p.time) ? p.time : null,
    moods: Array.isArray(p.moods) ? MOODS.filter((m) => p.moods.includes(m.id)).map((m) => m.id) : [],
  }
}

export function loadState(storage) {
  let raw
  try {
    raw = storage.getItem(KEY)
  } catch {
    return defaultState()
  }
  if (raw == null) return defaultState()
  try {
    const data = JSON.parse(raw)
    if (!data || !Array.isArray(data.items)) throw new Error('shape')
    return { items: data.items.map(cleanItem).filter(Boolean), prefs: cleanPrefs(data.prefs ?? {}) }
  } catch {
    try {
      storage.setItem(`${KEY}:broken`, raw)
    } catch {
      // 退避できなくても起動は続ける
    }
    return defaultState()
  }
}

export function saveState(storage, state) {
  try {
    storage.setItem(KEY, JSON.stringify({ items: state.items, prefs: state.prefs }))
    return true
  } catch {
    return false
  }
}
```
